SELECT CONCAT(fn.name, ' ', ln.name) AS fullname,p.birthdate,p.nationality,GROUP_CONCAT(p.playerid ORDER BY p.playerid) AS ids,COUNT(*) AS count_ids
FROM players p LEFT JOIN playernames fn ON p.firstnameid = fn.nameid LEFT JOIN playernames ln ON p.lastnameid = ln.nameid WHERE ln.name NOT LIKE '%new player%'
GROUP BY fn.name, ln.name, p.birthdate, p.nationality HAVING COUNT(*) > 1 ORDER BY fullname;

Joueur dans plusieurs clubs:
SELECT p.playerid,CONCAT(fn.name, ' ', ln.name) AS fullname, GROUP_CONCAT(DISTINCT t.teamname ORDER BY t.teamname) AS clubs,COUNT(DISTINCT t.teamid) AS club_count
FROM teamplayerlinks tpl JOIN players p ON tpl.playerid = p.playerid LEFT JOIN playernames fn ON p.firstnameid = fn.nameid 
LEFT JOIN playernames ln ON p.lastnameid = ln.nameid JOIN teams t ON tpl.teamid = t.teamid JOIN leagueteamlinks ltl ON tpl.teamid = ltl.teamid
WHERE ltl.leagueid NOT IN (78) AND ( t.teamname NOT LIKE '%adidas%'AND t.teamname NOT LIKE '%world%' AND t.teamname NOT LIKE '%All stars%')
GROUP BY p.playerid, fullname HAVING COUNT(DISTINCT t.teamid) > 1 ORDER BY fullname;

exporter joueur dans plusieurs clubs en csv:

SELECT x.playerid,CONCAT(fn.name, ' ', ln.name) AS fullname,t1.teamname AS club1,t2.teamname AS club2,x.teamid1,x.teamid2
FROM ( SELECT tpl.playerid,MIN(t.teamid) AS teamid1,MAX(t.teamid) AS teamid2 FROM teamplayerlinks tpl
JOIN teams t ON tpl.teamid = t.teamid JOIN leagueteamlinks ltl ON tpl.teamid = ltl.teamid
WHERE ltl.leagueid <> 78 AND t.teamname NOT LIKE '%adidas%' AND t.teamname NOT LIKE '%world%' AND t.teamname NOT LIKE '%All stars%'
GROUP BY tpl.playerid HAVING COUNT(DISTINCT t.teamid) = 2) AS x 
JOIN players p ON x.playerid = p.playerid LEFT JOIN playernames fn ON p.firstnameid = fn.nameid
LEFT JOIN playernames ln ON p.lastnameid = ln.nameid JOIN teams t1 ON t1.teamid = x.teamid1 JOIN teams t2 ON t2.teamid = x.teamid2
ORDER BY fullname INTO OUTFILE '/tmp/export.csv' FIELDS TERMINATED BY ';' OPTIONALLY ENCLOSED BY '"' LINES TERMINATED BY '\n';

UPDATE players p
JOIN (
    SELECT 
        name_bin,
        MAX(nameid) AS keep_id,
        GROUP_CONCAT(nameid ORDER BY nameid) AS all_ids
    FROM (
        SELECT 
            name COLLATE utf8mb4_bin AS name_bin,
            nameid
        FROM playernames
    ) AS x
    GROUP BY name_bin
    HAVING COUNT(*) > 1
) d
ON FIND_IN_SET(p.firstnameid, d.all_ids)
   OR FIND_IN_SET(p.lastnameid, d.all_ids)
   OR FIND_IN_SET(p.commonnameid, d.all_ids)
SET 
    p.firstnameid  = IF(FIND_IN_SET(p.firstnameid, d.all_ids), d.keep_id, p.firstnameid),
    p.lastnameid   = IF(FIND_IN_SET(p.lastnameid, d.all_ids), d.keep_id, p.lastnameid),
    p.commonnameid = IF(FIND_IN_SET(p.commonnameid, d.all_ids), d.keep_id, p.commonnameid);
DELETE pn
FROM playernames pn
JOIN (
    SELECT 
        name_bin,
        MAX(nameid) AS keep_id,
        GROUP_CONCAT(nameid ORDER BY nameid) AS all_ids
    FROM (
        SELECT 
            name COLLATE utf8mb4_bin AS name_bin,
            nameid
        FROM playernames
    ) AS x
    GROUP BY name_bin
    HAVING COUNT(*) > 1
) d ON FIND_IN_SET(pn.nameid, d.all_ids)
WHERE pn.nameid <> d.keep_id;
